import type { WeightConfig } from './types'

// Default weights when user hasn't answered questions
export const DEFAULT_WEIGHTS: WeightConfig = {
  integrationFit: 0.20,
  complianceMatch: 0.20,
  budgetFit: 0.20,
  featureMatch: 0.20,
  stackCompatibility: 0.20,
}

interface AssessmentInput {
  integrationNeeds?: string[]
  complianceRequirements?: string[]
  budgetRange?: string
  primaryUseCases?: string[]
  currentStack?: string[]
}

export function deriveWeights(assessment: AssessmentInput): WeightConfig {
  // Start with base weights
  const weights = {
    integrationFit: 0.15,
    complianceMatch: 0.15,
    budgetFit: 0.20,
    featureMatch: 0.25,
    stackCompatibility: 0.15,
  }

  // Increase weights based on user priorities
  if (assessment.integrationNeeds && assessment.integrationNeeds.length > 3) {
    weights.integrationFit += 0.10
  }
  if (assessment.complianceRequirements && assessment.complianceRequirements.length > 0) {
    weights.complianceMatch += 0.10
  }
  if (assessment.budgetRange === 'under-1000' || assessment.budgetRange === 'tight') {
    weights.budgetFit += 0.10
  }
  if (assessment.currentStack && assessment.currentStack.length > 2) {
    weights.stackCompatibility += 0.10
  }

  // Cap each weight at 0.35
  Object.keys(weights).forEach(key => {
    weights[key as keyof typeof weights] = Math.min(weights[key as keyof typeof weights], 0.35)
  })

  // Normalize to sum to 1.0
  const sum = Object.values(weights).reduce((a, b) => a + b, 0)
  Object.keys(weights).forEach(key => {
    weights[key as keyof typeof weights] /= sum
  })

  return weights
}

import type { Criterion, PlatformScore, ScoringContext, AuditEntry } from './types'
import { normalizeMinMax, CRITERION_DIRECTIONS } from './normalize'

export function calculateSAW(criteria: Criterion[]): number {
  if (criteria.length === 0) return 0
  const weightedSum = criteria.reduce(
    (sum, c) => sum + (c.weight * c.normalizedValue),
    0
  )
  return Math.round(weightedSum * 100)
}

export function scorePlatform(platform: Platform, context: ScoringContext): PlatformScore {
  // Build criteria array with normalized values
  const criteria: Criterion[] = [
    buildCriterion('integrationFit', platform, context),
    buildCriterion('complianceMatch', platform, context),
    buildCriterion('budgetFit', platform, context),
    buildCriterion('featureMatch', platform, context),
    buildCriterion('stackCompatibility', platform, context),
  ]

  const totalScore = calculateSAW(criteria)
  const auditTrail = generateAuditTrail(platform, criteria)

  return {
    platformId: platform.slug,
    platformName: platform.title,
    totalScore,
    criteriaScores: criteria,
    auditTrail,
  }
}

export function scoreAllPlatforms(platforms: Platform[], context: ScoringContext): PlatformScore[] {
  return platforms
    .map(p => scorePlatform(p, context))
    .sort((a, b) => b.totalScore - a.totalScore)
}
