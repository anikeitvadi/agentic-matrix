import type { AuditEntry, PlatformScore, Criterion } from './types'

/**
 * Generate human-readable audit trail for a platform's score
 */
export function generateAuditTrail(
  platformName: string,
  criteria: Criterion[]
): AuditEntry[] {
  return criteria.map(criterion => ({
    criterionName: criterion.name,
    rawValue: criterion.value,
    normalizedValue: criterion.normalizedValue,
    weight: criterion.weight,
    weightedScore: criterion.weight * criterion.normalizedValue,
    reasoning: formatCriterionReasoning(criterion),
  }))
}

/**
 * Format a single criterion's contribution to the score
 */
function formatCriterionReasoning(criterion: Criterion): string {
  const weightPercent = Math.round(criterion.weight * 100)
  const normalizedPercent = Math.round(criterion.normalizedValue * 100)
  const contribution = Math.round(criterion.weight * criterion.normalizedValue * 100)

  const criterionLabels: Record<string, string> = {
    integrationFit: 'Integration compatibility',
    complianceMatch: 'Compliance coverage',
    budgetFit: 'Budget alignment',
    featureMatch: 'Feature coverage',
    stackCompatibility: 'Stack compatibility',
  }

  const label = criterionLabels[criterion.name] || criterion.name

  return `${label}: Scored ${normalizedPercent}% (contributes ${contribution} points at ${weightPercent}% weight)`
}

/**
 * Explain why one platform scored higher than another
 */
export function explainComparison(
  winner: PlatformScore,
  loser: PlatformScore
): string {
  const scoreDiff = winner.totalScore - loser.totalScore

  // Find criteria where winner outperformed loser
  const advantages = winner.criteriaScores
    .map((winnerCriterion, i) => {
      const loserCriterion = loser.criteriaScores[i]
      const winnerContribution = winnerCriterion.weight * winnerCriterion.normalizedValue
      const loserContribution = loserCriterion.weight * loserCriterion.normalizedValue
      return {
        name: winnerCriterion.name,
        diff: (winnerContribution - loserContribution) * 100,
        winnerScore: Math.round(winnerCriterion.normalizedValue * 100),
        loserScore: Math.round(loserCriterion.normalizedValue * 100),
      }
    })
    .filter(a => a.diff > 0.5) // Meaningful differences only
    .sort((a, b) => b.diff - a.diff)

  const criterionLabels: Record<string, string> = {
    integrationFit: 'integration compatibility',
    complianceMatch: 'compliance coverage',
    budgetFit: 'budget alignment',
    featureMatch: 'feature coverage',
    stackCompatibility: 'stack compatibility',
  }

  if (advantages.length === 0) {
    return `${winner.platformName} and ${loser.platformName} scored similarly across all criteria.`
  }

  const topAdvantage = advantages[0]
  const label = criterionLabels[topAdvantage.name] || topAdvantage.name

  let explanation = `${winner.platformName} scores ${scoreDiff} points higher than ${loser.platformName}. `
  explanation += `The main advantage is ${label}: ${winner.platformName} scored ${topAdvantage.winnerScore}% vs ${loser.platformName}'s ${topAdvantage.loserScore}%.`

  if (advantages.length > 1) {
    const others = advantages.slice(1, 3).map(a => criterionLabels[a.name] || a.name)
    explanation += ` Other advantages: ${others.join(', ')}.`
  }

  return explanation
}

/**
 * Format an audit entry for display
 */
export function formatAuditEntry(entry: AuditEntry): string {
  return entry.reasoning
}

```typescript
'use client'

import { useState } from 'react'
import type { PlatformScore } from '@/lib/scoring/types'
import { explainComparison } from '@/lib/scoring/audit-trail'

interface Props {
  scores: PlatformScore[]
}

export function AuditTrail({ scores }: Props) {
  const [selectedPlatform, setSelectedPlatform] = useState<string | null>(
    scores[0]?.platformId ?? null
  )
  const [compareTo, setCompareTo] = useState<string | null>(
    scores[1]?.platformId ?? null
  )

  const selectedScore = scores.find(s => s.platformId === selectedPlatform)
  const compareScore = scores.find(s => s.platformId === compareTo)

  return (
    <div className="space-y-6">
      <h2 className="text-2xl font-semibold">Decision Audit Trail</h2>
      <p className="text-neutral-400">
        Understand exactly why each platform scored the way it did.
      </p>

      {/* Platform selector */}
      <div className="flex gap-4 flex-wrap">
        <div>
          <label className="block text-sm font-medium mb-1">View details for:</label>
          <select
            value={selectedPlatform ?? ''}
            onChange={(e) => setSelectedPlatform(e.target.value)}
            className="p-2 bg-neutral-800 rounded border border-neutral-700"
          >
            {scores.map(s => (
              <option key={s.platformId} value={s.platformId}>
                {s.platformName} ({s.totalScore} pts)
              </option>
            ))}
          </select>
        </div>

        <div>
          <label className="block text-sm font-medium mb-1">Compare to:</label>
          <select
            value={compareTo ?? ''}
            onChange={(e) => setCompareTo(e.target.value)}
            className="p-2 bg-neutral-800 rounded border border-neutral-700"
          >
            {scores.filter(s => s.platformId !== selectedPlatform).map(s => (
              <option key={s.platformId} value={s.platformId}>
                {s.platformName} ({s.totalScore} pts)
              </option>
            ))}
          </select>
        </div>
      </div>

      {/* Score breakdown */}
      {selectedScore && (
        <div className="p-4 bg-neutral-900 rounded-lg border border-neutral-800">
          <h3 className="font-semibold mb-3">
            {selectedScore.platformName} Score Breakdown
          </h3>
          <div className="space-y-2">
            {selectedScore.auditTrail.map((entry, i) => (
              <div key={i} className="flex justify-between text-sm">
                <span className="text-neutral-400">{entry.criterionName}</span>
                <span>
                  {Math.round(entry.normalizedValue * 100)}%
                  <span className="text-neutral-500 ml-2">
                    (+{Math.round(entry.weightedScore * 100)} pts)
                  </span>
                </span>
              </div>
            ))}
            <div className="border-t border-neutral-700 pt-2 flex justify-between font-semibold">
              <span>Total Score</span>
              <span className="text-brand-500">{selectedScore.totalScore}/100</span>
            </div>
          </div>
        </div>
      )}

      {/* Comparison explanation */}
      {selectedScore && compareScore && (
        <div className="p-4 bg-neutral-900 rounded-lg border border-brand-800">
          <h3 className="font-semibold mb-2 text-brand-400">
            Why {selectedScore.platformName} vs {compareScore.platformName}?
          </h3>
          <p className="text-neutral-300">
            {explainComparison(selectedScore, compareScore)}
          </p>
        </div>
      )}
    </div>
  )
}
```
