2. Create app/blueprints/[slug]/page.tsx (detail page):
```tsx
import { blueprints, platforms } from '.velite'
import { notFound } from 'next/navigation'
import Link from 'next/link'
import { BlueprintContent } from './BlueprintContent'

export function generateStaticParams() {
  return blueprints.map((blueprint) => ({
    slug: blueprint.slug,
  }))
}

export default async function BlueprintPage({
  params,
}: {
  params: Promise<{ slug: string }>
}) {
  const { slug } = await params
  const blueprint = blueprints.find((b) => b.slug === slug)

  if (!blueprint) {
    notFound()
  }

  // Get applicable platform details
  const applicablePlatformDetails = platforms.filter((p) =>
    blueprint.applicablePlatforms.includes(p.slug)
  )

  return (
    <main className="container mx-auto px-4 py-8 max-w-4xl">
      {/* Breadcrumb */}
      <nav className="mb-6 text-sm">
        <Link href="/blueprints" className="text-brand-600 hover:underline">
          Blueprints
        </Link>
        <span className="mx-2 text-neutral-400">/</span>
        <span className="text-neutral-600">{blueprint.title}</span>
      </nav>

      {/* Header */}
      <header className="mb-8">
        <h1 className="text-3xl font-bold text-neutral-900 mb-2">
          {blueprint.title}
        </h1>
        <p className="text-lg text-neutral-600 mb-4">
          {blueprint.description}
        </p>

        {/* Metadata grid */}
        <div className="grid grid-cols-2 md:grid-cols-4 gap-4 p-4 bg-neutral-50 rounded-lg">
          <div>
            <p className="text-xs text-neutral-500 uppercase">Complexity</p>
            <p className={`font-medium ${
              blueprint.complexity === 'simple' ? 'text-green-700' :
              blueprint.complexity === 'moderate' ? 'text-amber-700' :
              'text-red-700'
            }`}>
              {blueprint.complexity}
            </p>
          </div>
          <div>
            <p className="text-xs text-neutral-500 uppercase">Build Time</p>
            <p className="font-medium text-neutral-900">
              {blueprint.estimatedDuration.build}
            </p>
          </div>
          <div>
            <p className="text-xs text-neutral-500 uppercase">Total Duration</p>
            <p className="font-medium text-neutral-900">
              {blueprint.estimatedDuration.foundation} + {blueprint.estimatedDuration.build} + {blueprint.estimatedDuration.test} + {blueprint.estimatedDuration.deploy}
            </p>
          </div>
          <div>
            <p className="text-xs text-neutral-500 uppercase">Last Verified</p>
            <p className="font-medium text-neutral-900">
              {new Date(blueprint.lastVerified).toLocaleDateString()}
            </p>
          </div>
        </div>
      </header>

      {/* Applicable platforms */}
      <section className="mb-8">
        <h2 className="text-lg font-semibold text-neutral-900 mb-3">
          Applicable Platforms
        </h2>
        <div className="flex flex-wrap gap-2">
          {applicablePlatformDetails.map((platform) => (
            <Link
              key={platform.slug}
              href={`/platforms/${platform.slug}`}
              className={`px-3 py-1 text-sm rounded-full border ${
                blueprint.recommendedPlatforms?.includes(platform.slug)
                  ? 'bg-brand-50 border-brand-300 text-brand-800'
                  : 'bg-neutral-50 border-neutral-200 text-neutral-700'
              } hover:border-brand-500`}
            >
              {platform.title}
              {blueprint.recommendedPlatforms?.includes(platform.slug) && (
                <span className="ml-1 text-xs">(recommended)</span>
              )}
            </Link>
          ))}
        </div>
      </section>

      {/* Prerequisites */}
      {blueprint.prerequisites.length > 0 && (
        <section className="mb-8 p-4 bg-amber-50 border border-amber-200 rounded-lg">
          <h2 className="text-lg font-semibold text-amber-900 mb-2">
            Prerequisites
          </h2>
          <ul className="list-disc list-inside text-sm text-amber-800 space-y-1">
            {blueprint.prerequisites.map((prereq, i) => (
              <li key={i}>{prereq}</li>
            ))}
          </ul>
        </section>
      )}

      {/* MDX Content */}
      <article className="prose prose-neutral max-w-none">
        <BlueprintContent code={blueprint.body} />
      </article>
    </main>
  )
}
```

3. Create app/blueprints/[slug]/BlueprintContent.tsx (client component for MDX):
```tsx
'use client'

import { useMDXComponent } from 'next-contentlayer/hooks'
import { Admonition } from '@/components/ui/Admonition'
import { ImplementationChecklist, ChecklistPhase, ChecklistItem } from '@/components/blueprint/ImplementationChecklist'
import { PlatformCallout } from '@/components/blueprint/PlatformCallout'

// Note: For Velite MDX, we need to use a different approach than Contentlayer
// The MDX is pre-compiled, so we evaluate it directly

interface BlueprintContentProps {
  code: string
}

export function BlueprintContent({ code }: BlueprintContentProps) {
  // Velite compiles MDX to a function body string
  // We need to evaluate it with the component context
  const Component = useMDXComponent(code)

  return (
    <Component
      components={{
        Admonition,
        ImplementationChecklist,
        ChecklistPhase,
        ChecklistItem,
        PlatformCallout,
      }}
    />
  )
}
```

**Note on MDX rendering:** Velite with s.mdx() compiles MDX differently than Contentlayer. If `useMDXComponent` doesn't work, check Velite docs for the correct MDX rendering approach. May need to use `getMDXComponent` from Velite's runtime or a custom MDX component loader.

**Fallback approach if useMDXComponent fails:**
- Check if Velite exports compiled MDX as React components directly
- May need to render the HTML content if MDX compilation provides HTML output
- Document the issue in SUMMARY if a workaround is needed
