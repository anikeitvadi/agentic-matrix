<Mermaid chart={`
sequenceDiagram
    participant User
    participant Agent

    participant Validator
    participant Review as Human Review
    participant DB as Database

    User->>Agent: Upload document
    Agent->>Parser: Extract raw text/fields
    Parser-->>Agent: Raw data + confidence
    Agent->>Validator: Validate against schema

        Validator-->>Agent: Valid data

        Agent->>DB: Store with corrections
        Agent-->>User: Confirmation after review
    end
`} />
