Create `lib/assessment/progress-storage.ts` with SSR-safe localStorage helpers.

The react-hook-form-persist library handles form data, but we need separate storage for:
- Current step number
- Assessment start timestamp (for analytics later)
- Any metadata not in the form

```typescript
const STORAGE_KEY = 'assessment-progress'
const STEP_KEY = 'assessment-current-step'

// SSR-safe check
function isClient(): boolean {
  return typeof window !== 'undefined'
}

// Save current step (separate from form data)
export function saveCurrentStep(step: number): void {
  if (!isClient()) return
  try {
    window.localStorage.setItem(STEP_KEY, String(step))
  } catch (e) {
    // localStorage might be full or disabled
    console.warn('Failed to save step to localStorage:', e)
  }
}

// Load current step with SSR-safe lazy initialization
export function loadCurrentStep(): number {
  if (!isClient()) return 1
  try {
    const saved = window.localStorage.getItem(STEP_KEY)
    return saved ? parseInt(saved, 10) : 1
  } catch (e) {
    return 1
  }
}

// Clear all progress (call on successful submission)
export function clearProgress(): void {
  if (!isClient()) return
  try {
    window.localStorage.removeItem(STORAGE_KEY)
    window.localStorage.removeItem(STEP_KEY)
  } catch (e) {
    console.warn('Failed to clear localStorage:', e)
  }
}

// Check if there's saved progress (for showing "Resume" UI)
export function hasSavedProgress(): boolean {
  if (!isClient()) return false
  try {
    return window.localStorage.getItem(STORAGE_KEY) !== null
  } catch (e) {
    return false
  }
}

// Get the storage key for react-hook-form-persist
export function getFormStorageKey(): string {
  return STORAGE_KEY
}
```

All functions handle:
- SSR (return safe defaults when window undefined)
- localStorage errors (quota exceeded, disabled in private browsing)
