```typescript
/**
 * Format currency values with appropriate precision
 * @param value - Dollar amount
 * @param options - compact: use K/M suffixes, showCents: include decimals
 */
export function formatCurrency(
  value: number,
  options: { compact?: boolean; showCents?: boolean } = {}
): string

/**
 * Format token counts with K/M/B suffixes
 */
export function formatTokenCount(tokens: number): string

/**
 * Format duration in days or weeks
 * @param days - Number of engineering days
 * @returns "X days" or "X weeks" for larger values
 */
export function formatDuration(days: number): string

/**
 * Format a range of days
 */
export function formatRange(low: number, high: number): string
```

Implementation details from research:
- compact mode: >= 1M shows "$1.2M", >= 1K shows "$45K"
- standard mode: use Intl.NumberFormat with 'en-US', 'USD'
- showCents: minimumFractionDigits: 2 for detailed views
- duration: >= 20 days converts to weeks (round to nearest week)

Use Intl.NumberFormat for locale-aware formatting (do NOT hand-roll number formatting).
